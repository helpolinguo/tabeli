%%K t16-01-1 p
1. --- اقتربت السنة الجديدة. وقد هيّأت المتاجر الكبيرة معارض
\VUgras{اللُّعَب} \textsuperscript{(1)} وهدايا رأس السنة.

%%K t16-01-2 p
وطلبت من جدي أن يأخذني إلى البازار لأرى هذا المعرض الجميل، فقبل.

%%K t16-02-1 p
2. --- وقفنا أولًا أمام لوحات سلاح جميلة عليها \VUgras{بندقية}
و\VUgras{قبعة عسكرية} و\VUgras{سيف} و\VUgras{حزام سيف} وزوج
\VUgras{كتفيات}، أو \VUgras{خوذة} و\VUgras{درع صدر} \textsuperscript{(3)}
و\VUgras{مسدس} \textsuperscript{(4)}. وكان ذلك كله يعنيني أكثر بكثير من
\VUgras{الفوانيس السحرية} \textsuperscript{(2)}.

\VUsaut{3.0mm}
\VUpk{10.2pt}{40}{مناظر جميلة.}

%%K t16-03-1 p
3. --- وفي القسم نفسه كان \VUgras{حارس} \textsuperscript{(5)} في \VUgras{كوخه}؛
وبدا كأنه يحرس حديقة حيوان كاملة من الورق المقوّى. وما أكثر ما كان فيها
من حيوان! \VUgras{ببغاء} \textsuperscript{(6)} على مجثمه، و\VUgras{كركدّن}
\textsuperscript{(7)} على أنفه قرن، و\VUgras{رَنّة} \textsuperscript{(8)}،
و\VUgras{نعامة} \textsuperscript{(9)}، و\VUgras{قرد} \textsuperscript{(10)} يكسر
جوزة، و\VUgras{ثعلب} \textsuperscript{(11)} يحمل دجاجة، و\VUgras{تمساح}
\textsuperscript{(12)} فاغر فكّيه الهائلين، و\VUgras{جمل} \textsuperscript{(13)}،
و\VUgras{سلوقي} \textsuperscript{(14)} أنيق، و\VUgras{فهد} \textsuperscript{(15)}، ثم
حيوانان لم أكن أعرفهما وهما، على ما قيل لي، \VUgras{ابن عرس}
\textsuperscript{(16)} و\VUgras{ضبع} \textsuperscript{(17)}. وكان هناك أيضًا
\VUgras{نمر مرقّط} \textsuperscript{(18)} و\VUgras{ذئب} \textsuperscript{(21)}؛ وعلى
مقربة \VUgras{جرذان} \textsuperscript{(19)} لطيفة و\VUgras{فئران} \textsuperscript{(20)}
صغيرة جدًا من المطاط. وأخيرًا أتمّ المجموعةَ \VUgras{فرس نهر}
\textsuperscript{(22)} و\VUgras{جمل وحيد السنام} \textsuperscript{(23)} أحدب. ولبثت
هناك ساعات لولا أن قربي مخيمًا بديعًا ملآن بـ\VUgras{جنود الرصاص}
\textsuperscript{(29)} استرعى انتباهي.

%%K t16-tit-2 sub
\VUsaut{4.0mm}
\VUpk{10.2pt}{40}{الجنود والحرب.}
\VUsaut{2.8mm}

%%K t16-04-1 p
4. --- وجدّي، وهو \VUgras{مُقعَد حرب} \textsuperscript{(24)} اضطر إلى ترك الجيش
بعد \VUgras{جرح} نال به \VUgras{الميدالية العسكرية}، يحب أن يحدّثني عن
\VUgras{الحملات} التي شارك فيها. وقد سرّه أن يشرح لي مختلف حركات ذلك
الجيش الصغير المصغّر. ووقفت قربنا جماعة من جنود حقيقيين يزورون البازار ---
\VUgras{مهندس عسكري} \textsuperscript{(25)}، و\VUgras{جندي جبلي} \textsuperscript{(26)}
في \VUgras{قبعته المستديرة}، و\VUgras{رقيب أول} \textsuperscript{(27)} مشاة،
و\VUgras{رقيب} \textsuperscript{(28)} مدفعية على كمّه \VUgras{شريط} ذهبي ---
لتسمع الشرح.

%%K t16-05-1 p
5. --- وجدّي، وقد فخر بجمهور بهذه المكانة، ارتجل خطة معركة كاملة.

%%K t16-05-2 p
كانت المدينة المحصَّنة، بـ\VUgras{أسوارها} و\VUgras{خنادقها}، محاصَرة
بـ\VUgras{جيش العدو} الذي كان يتأهب، بعد \VUgras{قصف} طويل، لاقتحامها
\VUgras{اقتحامًا}. لكن \VUgras{المحاصَرين}، وقد دفعهم الجوع، قاموا
بـ\VUgras{هجمة} على \VUgras{مخيم} \VUgras{المحاصِرين}. وبعد أن ردّوا
\VUgras{الهجوم} في \VUgras{قتال} ضارٍ حول \VUgras{الخيام}، أرسل المحاصِرون
\VUgras{المنتصرون} \VUgras{أسرابهم} في \VUgras{مطاردة} المهاجمين
\VUgras{المهزومين}. فـ\VUgras{تراجع} هؤلاء تاركين \VUgras{ميدان المعركة}
مفروشًا بـ\VUgras{القتلى} و\VUgras{الجرحى}. وحمل \VUgras{حاملو النقّالات}
الجرحى إلى \VUgras{المستشفى الميداني} ليعالجهم \VUgras{الجرّاح}.

%%K t16-05-3 p
ورغم المقاومة البطولية لبضع \VUgras{كتائب} انتظمت في \VUgras{مربّع}، كانت
\VUgras{الهزيمة} تامة؛ و\VUgras{فرّ} ما بقي من الجيش تاركًا وراءه
\VUgras{أسرى} كثيرين. وكاد العدو يتوّج \VUgras{انتصاره} بأخذ المدينة. ولعل
تلك نهاية الحملة، وبعد \VUgras{هدنة} يمكن أن تُوقَّع \VUgras{معاهدة سلام}.

%%K t16-tit-3 sub
\VUsaut{4.4mm}
\VUpk{10.2pt}{40}{هدايا رأس السنة.}
\VUsaut{2.8mm}

%%K t16-06-1 p
6. --- والجنود الشبان، وقد ضحكوا وهم يسمعون حكاية المحارب القديم الملوّنة،
سألوه أسئلة أخرى. وأما أنا فدرت حول المِنضدة لأنظر في \VUgras{قوس نشّاب}
\textsuperscript{(32)} جميلة و\VUgras{قوس} \textsuperscript{(33)} مع \VUgras{سهمها}
\textsuperscript{(31)}. فوجدت هناك مجموعة أخرى من الحيوانات:
\VUgras{طائرين مغرّدين} \textsuperscript{(34)}، و\VUgras{بلبلًا} و\VUgras{هازجة}
بزنبرك يغنّيان بأعلى صوتيهما، وسلسلة من حيوانات غريبة ---
\VUgras{خفّاش} \textsuperscript{(35)}، و\VUgras{ثعبان بوا} \textsuperscript{(36)}،
و\VUgras{سلحفاة} \textsuperscript{(37)}، و\VUgras{فقمة} \textsuperscript{(38)}،
و\VUgras{حوت} \textsuperscript{(39)}، و\VUgras{قرش} \textsuperscript{(40)} --- من
جلد رقيق.

%%K t16-07-1 p
7. --- ولم أقف طويلًا لأنظر إليها، فقد لمحت ما كنت أحلم به:
\VUgras{مسرح عرائس} \textsuperscript{(41)} رائع بـ\VUgras{مناظر} بديعة
و\VUgras{ستارة} حمراء بهيجة. وعلى \VUgras{الخشبة} بدا ممثلان يؤديان
\VUgras{دورًا} في \VUgras{مسرحية} شديدة الطرافة، فـ\VUgras{المتفرجون}
الصغار من الورق المقوّى في المقصورات، أو الجالسون في \VUgras{الصالة} وفي
\VUgras{الشرفة} خلف \VUgras{قائد الأوركسترا}، كانوا يصفّقون بحرارة.

%%K t16-07-2 p
لكن ذلك المسرح، للأسف، كان غالي الثمن جدًا.

%%K t16-08-1 p
8. --- ثم إن الاختيار بين اللُّعَب كان عسيرًا، فالواجهات مكدّسة بلعب من كل
نوع: \VUgras{هرلكان} \textsuperscript{(42)}، و\VUgras{بولشينيلا} \textsuperscript{(43)}،
و\VUgras{بييرو} \textsuperscript{(44)}، و\VUgras{بوم} \textsuperscript{(45)} يرفرف
بأجنحته، و\VUgras{شحارير} \textsuperscript{(46)} تصفر، و\VUgras{كلاب بودل} تنبح،
و\VUgras{غرامافون} \textsuperscript{(47)}، و\VUgras{ألغاز تركيب}. وقد أضحكتني
إحدى هذه اللعب كثيرًا: كانت تمثّل \VUgras{معركة بحرية} \textsuperscript{(48)}
تُهاجَم فيها \VUgras{سفينة} من \VUgras{قراصنة} قبالة شاطئ مغروس
بـ\VUgras{نخيل جوز الهند}.

9. --- وقد استطعت أن أنظر في هذه العجائب كلها على مهلي، إذ لم يكن في
المتجر إلا قليل من الناس --- بالكاد حفنة من المتسكعين، و\VUgras{فارس}
\textsuperscript{(49)}، و\VUgras{محصّل} \textsuperscript{(50)} يقدّم
\VUgras{كمبيالة} عند \VUgras{الصندوق} \textsuperscript{(51)} الرئيسي.

%%K t16-10-1 p
10. --- وسألت جدي عمّا تفيده الكتب على الرفوف خلف \VUgras{أمين الصندوق}؛
فقال لي إنها \VUgras{دفاتر الحسابات}.

11. --- ولما تركنا قسم اللُّعَب ذهبنا إلى قسم السروج لنلقي نظرة على
\VUgras{السروج} \textsuperscript{(53)} و\VUgras{الرِّكابات} \textsuperscript{(54)}
و\VUgras{الأعنّة} \textsuperscript{(55)} و\VUgras{اللُّجُم} \textsuperscript{(56)}
و\VUgras{السياط}. وبينما كان \VUgras{رئيس القسم} \textsuperscript{(57)} يعطي جدي
بعض المعلومات، ذهبت أنظر في معرض \VUgras{الخزف} و\VUgras{الزجاجيات}
\textsuperscript{(58)} حيث كانت \VUgras{سلطانيات} جميلة و\VUgras{أباريق شاي}
\textsuperscript{(59)} رقيقة.

%%K t16-12-1 p
12. --- ولنصعد إلى الطابق الأول مررنا خلف واجهة \VUgras{المشدّات}
\textsuperscript{(60)}، إلى جانب قسم الجوارب، حيث بُسطت \VUgras{سراويل داخلية}
\textsuperscript{(63)} جميلة جدًا. وقربها كانت \VUgras{بائعة} \textsuperscript{(61)}

\textsuperscript{(64)} حريرية جميلة.

%%K t16-12-2 p

يابانية و\VUgras{مظلات} \textsuperscript{(66)} و\VUgras{نخيل} \textsuperscript{(70)}
هائل.

%%K t16-13-1 p
13. --- ولم يكن في الطابق الأول كثير مما يُرى: أثاث فحسب (*)،
و\VUgras{خزائن حديدية} \textsuperscript{(67)}، و\VUgras{خزائن أدراج}
\textsuperscript{(68)}، و\VUgras{عربات أطفال} \textsuperscript{(69)}. لكن منظر المتجر

%%K t16-14-1 p

\textsuperscript{(78)} يساوم \VUgras{البائع} \textsuperscript{(79)} على حافظة أوراق.

%%K t16-14-2 p
