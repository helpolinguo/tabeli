%%K t12-01-1 p
1. --- میں نے ہُݨے جرمنی دا اک سفر کیتا اے۔ نکلݨ توں پہلاں میں گڈیاں دے چھُٹّݨ دا
ویلا جاݨن لئی اک \VUgras{ویلے دی سُوچی} \textsuperscript{(15)} خریدی سی۔ ایہہ ویکھ
کے پئی سب توں سوکھی گڈی پیرس توں رات نو وجے چھُٹدی اے تے سٹراسبورگ اک وج کے تیرہ
منٹ اُتے اپڑدی اے، میں اپݨے سفر دی تیاری کیتی، تے اگلے دن اسٹیشن تیکر گڈی کرا لئی۔

%%K t12-02-1 p
2. --- جدوں میں ٹُرن دے وڈے کمرے وچ وڑیا، اک بُڈھے پنڈ دے \VUgras{پادری}
\textsuperscript{(2)} دے پچھے، جیہڑا ہتھ وچ اپݨا \VUgras{سفری تھیلا}
\textsuperscript{(3)} چُکی سی، تدوں تیکر بہت سارے \VUgras{مسافر} \textsuperscript{(1)}
آ چُکے سن۔

%%K t12-02-2 p
مینوں اک \VUgras{تار} \textsuperscript{(5)} گھلݨی سی، ایسے لئی میں اک
\VUgras{نگران} \textsuperscript{(6)} کولوں \VUgras{تار گھر} \textsuperscript{(4)} پُچھ
لیا۔

%%K t12-03-1 p
3. --- \VUgras{اُڈیک کمرے} \textsuperscript{(7)} وچ جاوݨ توں پہلاں میں آوݨ جاوݨ دا
اک \VUgras{ٹکٹ} \textsuperscript{(9)} لیݨ گیا۔

%%K t12-04-1 p
4. --- \VUgras{ٹکٹ کھڑکی} \textsuperscript{(8)} اُتے مینوں دیر نہ لگی، کیوں جے اوتھے
بھیڑ گھٹ سی۔ چھٹی اُتے جاندے اک \VUgras{سپاہی} \textsuperscript{(10)} نے مہربانی کر
کے مینوں اپݨی واری دے دِتی، سو مینوں اِنّا ویلا مل گیا پئی
\VUgras{اسٹیشن ماسٹر} \textsuperscript{(13)} کولوں کجھ گلاں پُچھ لواں، جیہڑے
\VUgras{پولیس افسر} \textsuperscript{(12)} تے ڈیوٹی اُتے کھڑے \VUgras{ژندارم}
\textsuperscript{(11)} نال گلاں کر رہے سن۔

%%K t12-tit-1 sub
\VUpk{10.2pt}{40}{سامان دی بکنگ۔}
\VUsaut{3.0mm}

%%K t12-05-1 p
5. --- \VUgras{کتاباں دے کھوکھے} \textsuperscript{(14)} توں اک اخبار خرید کے میں
اپݨا \VUgras{سامان} \textsuperscript{(17)} تُلوایا — اک \VUgras{قُلی}
\textsuperscript{(16)} اوہنوں ہُݨے اک \VUgras{سامان دی ریڑھی} \textsuperscript{(19)}
اُتے لیایا سی۔ \VUgras{بابو} \textsuperscript{(22)}، جیہڑے \VUgras{تول جنتر}
\textsuperscript{(21)} دے انچارج نیں، اوہناں مینوں دسیا پئی میرا صندوق پینتی کلو دا
اے، جیدے نال پنج کلو دا \VUgras{ودھاء} بݨدا اے، جیدے لئی مینوں
\VUgras{سامان دی کھڑکی} \textsuperscript{(23)} اُتے ودھیرا محصول دیݨا پوے گا۔

%%K t12-06-1 p
6. --- کیوں جے میں ویلے توں پہلاں سی، مینوں اسٹیشن وچ مسافراں دی چہل پہل ویکھݨ دا
ویلا مل گیا۔ کجھ \VUgras{امانت گھر} \textsuperscript{(25)} وچ نِکے تھیلے رکھ یا لے
رہے سن؛ دوجے \VUgras{ویلے دیاں سُوچیاں} \textsuperscript{(26)} ویکھ رہے سن۔ آندے
مسافر \VUgras{نکاس} \textsuperscript{(32)} ول کاہلی کردے سن، ہتھ وچ اپݨے
\VUgras{اٹیچی} \textsuperscript{(24)} چُکی۔ کجھ \VUgras{سامان دی سپردگی}
\textsuperscript{(27)} اُتے اُڈیک کردے تے اپݨی \VUgras{رسید} \textsuperscript{(29)}
وکھاندے، اپݨے صندوق، \VUgras{پیٹیاں} \textsuperscript{(28)} یا \VUgras{ٹوکریاں}
\textsuperscript{(30)} لیݨ لئی۔

%%K t12-07-1 p
7. --- \VUgras{چنگی دے کارندے} \textsuperscript{(33)} دھیان نال گٹھڑیاں پرکھدے سن پئی
کِتے کجھ دسّݨ جوگ تے نئیں۔

%%K t12-08-1 p
8. --- کجھ مسافر \VUgras{کھاݨ پیݨ دے کمرے} \textsuperscript{(34)} ول گئے، جِتھے اک
\VUgras{بیرا} \textsuperscript{(35)} اوہناں نوں پھرتی نال وَرتا رہیا سی۔ اوہناں نوں
کاہلی کرنی سی، کیوں جے \VUgras{گڈی} \textsuperscript{(36)}، جیہڑی ایکسپریس سی،
اسٹیشن اُتے ودھ دیر نئیں رُکدی۔

%%K t12-09-1 p
9. --- فیر میں ٹُرن دے پلیٹ فارم اُتے گیا تے اک سِدھا \VUgras{ڈبہ}
\textsuperscript{(37)} لبھیا، تاں جو راہ وچ بدلݨا نہ پوے۔ میں لانگھے والی اک گڈی وچ
چڑھیا، جیدے وچ سگرٹ پیݨ والیاں لئی اک \VUgras{کوٹھڑی} \textsuperscript{(38)} اے تے
اک « کلیاں بیبیاں » لئی رکھی ہوئی۔

\VUpk{10.2pt}{40}{رات دا ٹُرنا۔}

%%K t12-10-1 p
10. --- \VUgras{پائیدان} \textsuperscript{(40)} چڑھ کے، \VUgras{دروازہ}
\textsuperscript{(39)} بند کر کے، \VUgras{پردہ} \textsuperscript{(41)} چاڑھ کے تے اپݨے
نِکے تھیلے \VUgras{سامان دی جالی} \textsuperscript{(43)} وچ رکھ کے میں \VUgras{سیٹ}
\textsuperscript{(42)} اُتے بہہ گیا۔ \VUgras{لیمپ والے} \textsuperscript{(44)} نوں
بتیاں بالدے ویکھ کے مینوں یاد آیا پئی مینوں اک رات ڈبے وچ لنگھاؤݨی اے، تے میں
\VUgras{سِرہاݨے} \textsuperscript{(45)} کرائے اُتے دیݨ والے کارندے نوں سد کے اک منگ
لیا۔

%%K t12-11-1 p
11. --- گڈی دا کنڈیکٹر ٹکٹ پرکھݨ آیا۔ میں اوہدے کولوں پُچھیا پئی اسیں ٹُرے کیوں
نئیں، جدوں کہ ویلا ہو چُکیا اے۔ اوہنے جواب دِتا پئی \VUgras{گڈی دے مکھی}
\textsuperscript{(46)} \VUgras{سامان دے ڈبے} \textsuperscript{(47)} وچ سائیکلاں
رکھواؤݨ وچ لگے نیں۔ ایہدے توں بغیر، سارے \VUgras{پیر نیمے کرن دے ڈبے}
\textsuperscript{(48)} ہالے بدلے وی نئیں گئے۔

%%K t12-12-1 p
12. --- پر اسیں چھیتی ای ٹُرن والے سی، کیوں جے \VUgras{اگ والا}
\textsuperscript{(50)}، اپݨے \VUgras{کوئلے دے ڈبے} \textsuperscript{(49)} اُتے چڑھیا،
\VUgras{انجن} \textsuperscript{(54)} دی اگ تیز کرن نوں کوئلہ لے رہیا سی، تے
\VUgras{ڈرائیور} \textsuperscript{(51)} نِرا \VUgras{نائب اسٹیشن ماسٹر}
\textsuperscript{(52)} دے اشارے دی اُڈیک وچ سی، جیہڑے \VUgras{پلیٹ فارم}
\textsuperscript{(53)} اُتے کھڑے سن۔

%%K t12-13-1 p
13. --- انجن دا \VUgras{بوائلر} \textsuperscript{(56)} ہولے سُر وچ گرجدا سی؛
\VUgras{چمنی} \textsuperscript{(55)} \VUgras{بھاپ} \textsuperscript{(60)} رلے دھوئیں
دیاں دھاراں اُگلدی سی۔ اک تِکھی \VUgras{سیٹی} \textsuperscript{(59)} گونجی تے
\VUgras{پسٹن} \textsuperscript{(58)} چلݨ لگے، جیہڑے اوس بھارے جنتر دے \VUgras{پہیے}
\textsuperscript{(57)} گھماندے نیں۔

14. --- گڈی، جیہڑی \VUgras{پٹڑیاں} \textsuperscript{(64)} اُتے دوڑ رہی سی، تیز ہوندی
گئی؛ \VUgras{پلیٹ فارم} \textsuperscript{(65)} مُک گئے؛ اسیں \VUgras{پاخانیاں}
\textsuperscript{(66)} کولوں لنگھے، تے اک دوجی \VUgras{پٹڑی} \textsuperscript{(63)}

لنگھیا۔ چھپراں دے تھلے کارندے ہولی گڈی نال گھلیا \VUgras{مال} \textsuperscript{(72)}

\textsuperscript{(80)} بݨاؤݨ نوں۔

%%K t12-16-1 p
16. --- اسیں چھیکرلے \VUgras{کانٹے} \textsuperscript{(78)} کولوں لنگھے، جیدے کول
کانٹے والا کھڑا سی؛ اسیں \VUgras{اشارے دا چکر} \textsuperscript{(86)} پچھے چھڈیا تے
اسٹیشن دور وچ گُم ہو گیا۔

%%K t12-17-1 p
17. --- چھیتی ای اسیں اوس \VUgras{لوہے دے پُل} \textsuperscript{(74)} اُتے آ گئے
جیہڑا \VUgras{دریا} \textsuperscript{(81)} لنگھدا اے۔ پُل لنگھ کے اسیں دریا دے نال
نال ٹُرے، جیدے اُتے تکڑیاں \VUgras{کھِچݨ والیاں کشتیاں} \textsuperscript{(82)}

\VUgras{مسافر جہاز} \textsuperscript{(84)} وی ویکھیا، ٹھیک اوس ویلے جدوں اوہ اک
\VUgras{تردے گھاٹ} \textsuperscript{(85)} نال لگݨ والا سی۔

وچوں لنگھے تے اپݨے پچھے \VUgras{پھاٹک دے رکھوالیاں} دیاں بے شمار \VUgras{کوٹھڑیاں}

گئی۔

19. --- مینوں اچانک اک کارندے دی آواز نے جگا دِتا، جیہڑا کُوک رہیا سی:
« ڈوئچ آورکور! \textsuperscript{(*)} سارے اُترو! » ایہہ چنگی دی پڑتال سی تے سرحد
دیاں ساریاں جھنجٹ بھریاں رسماں۔ مینوں اپݨی گھڑی اسٹیشن دی \VUgras{گھڑی}
\textsuperscript{(89)} نال ملاؤݨی پئی، جیہڑی پنجاہ پنج منٹ اگے سی؛ مشکل نال جاگیا
ہویا میں \VUgras{وڈی سُوئی} \textsuperscript{(90)} تے \VUgras{نِکی}
\textsuperscript{(91)} وچ فرق اوکھا ای کر سکدا سی؛ \VUgras{ڈائل} \textsuperscript{(92)}
تے سویر دی دھند نال بالکل دھندلا سی۔

\VUgras{اشتہاراں} \textsuperscript{(93)} نوں پڑھدا رہیا جیہڑے اسٹیشن دیاں کندھاں
