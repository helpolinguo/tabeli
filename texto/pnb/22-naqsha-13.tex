%%K t13-01-1 p
1. --- کل شام روایاں اپڑ کے میں \textit{محیط ہوٹل} وچ ٹھہریا، جیدی سفارش اک یار نے
کیتی سی۔

%%K t13-01-2 p
مینوں اک سوہݨا \VUgras{کمرہ} \textsuperscript{(2)} مِلیا، دوجی \VUgras{منزل}
\textsuperscript{(1)} اُتے، جِتھے میں بہت چنگی طرحاں سُتا (*)۔

%%K t13-02-1 p
2. --- اج سویرے اپݨا کمرہ چھڈدے ہوئے، جِتھے \VUgras{نوکراݨی} \textsuperscript{(3)}
میرا ناشتہ لے آئی سی، میں \VUgras{سگرٹ والے کمرے} \textsuperscript{(5)} وچ گیا،
جیہڑا \VUgras{پڑھݨ دے کمرے} \textsuperscript{(4)} دا کم وی دیندا اے، تاں جو
\VUgras{اخبار} \textsuperscript{(6)} ویکھ لواں، اک \VUgras{سگرٹ} \textsuperscript{(8)}
پیندے ہوئے۔ پڑھ چُکݨ اُتے میں \VUgras{لفٹ} \textsuperscript{(9)} توں تھلے اُتریا تے
شہر وچ ٹہلݨ نکل پیا۔

%%K t13-03-1 p
3. --- کیوں جے میں ہوٹل دے \VUgras{بابو} \textsuperscript{(12)} کولوں پُچھݨا بھُل گیا
سی پئی \VUgras{سانجھی میز} \textsuperscript{(11)} اُتے کھاݨا کیس ویلے وَرتایا جاندا
اے، میں چھیتی مُڑ آیا تے موقعے دا فائدہ چُک کے ہوٹل دے \VUgras{نہاؤݨ دے کمرے}
\textsuperscript{(10)} وچ نہا لیا۔ فیر میں اپݨے اک یار نوں فون کرن
\VUgras{گلّے} \textsuperscript{(15)} اُتے مُڑیا۔ پر \VUgras{ٹیلیفون}
\textsuperscript{(13)} رُجھیا ہویا سی؛ مینوں اُلجھݨ وچ ویکھ کے \VUgras{خزانچی}
\textsuperscript{(14)}، جیہڑی اک \VUgras{بل} \textsuperscript{(17)}
\VUgras{پراہُݨیاں دی وہی} \textsuperscript{(16)} وچ چاڑھ رہی سی، اوہنے
\VUgras{بُوہے دے رکھوالے} \textsuperscript{(18)} نوں آکھیا پئی \VUgras{چھوکرے}
\textsuperscript{(19)} نوں گھل دیوے، جیہڑا اپݨی \VUgras{سائیکل} \textsuperscript{(20)}
اُتے میرا کم کر آوے۔

%%K t13-04-1 p
4. --- میں اوہناں دا شکریہ کیتا تے باہر جا کے \VUgras{قُلی} \textsuperscript{(24)}
(پھُٹکل کم والے) نوں بخشش دِتی، جیہڑا اپݨی \VUgras{ہتھ ریڑھی} \textsuperscript{(25)}
اُتے اسٹیشن توں میرا \VUgras{ٹوپی دا بکس} \textsuperscript{(26)}، میرا
\VUgras{اٹیچی} \textsuperscript{(27)} تے میرا \VUgras{سفری تھیلا} \textsuperscript{(28)}
لیایا سی۔ \VUgras{ڈاکیا} \textsuperscript{(21)}، جیہڑا ٹھیک اوسے ویلے آیا، اوہنے
مینوں اک \VUgras{چِٹھی} \textsuperscript{(22)} پھڑائی۔

%%K t13-05-1 p
5. --- میں \VUgras{چِٹھیاں دی پیٹی} \textsuperscript{(23)} کول دہلیز اُتے کجھ دیر ہور
کھڑا رہیا، گلی دی چہل پہل ویکھدا۔ \VUgras{چالک} \textsuperscript{(30)}، جیہڑا
\VUgras{اومنی بس} \textsuperscript{(29)} دا اے، اپݨی گڈی اُتے اک \VUgras{صندوق}
\textsuperscript{(31)} لد رہیا سی، جیہڑا اک \VUgras{پراہُݨے} \textsuperscript{(32)} دا
سی، جیدے توں \VUgras{ہوٹل والا} \textsuperscript{(33)} ودا لے رہیا سی۔ اک جوان
\VUgras{تار والا مُنڈا} \textsuperscript{(35)} بغیر کِسے کاہلی دے ہوٹل دے اک پراہُݨے
لئی اک \VUgras{تار} \textsuperscript{(34)} لیا رہیا سی؛ اک \VUgras{سیلانی}
\textsuperscript{(36)}، مونڈھے اُتے اپݨا \VUgras{کیمرہ} \textsuperscript{(38)} لٹکائی،
اپݨی \VUgras{راہ دسیری} \textsuperscript{(37)} ویکھ رہیا سی۔

%%K t13-tit-1 sub
\VUsaut{4.0mm}
\VUpk{10.2pt}{40}{دوپہر دے کھاݨے دا ویلا۔}
\VUsaut{3.0mm}

%%K t13-06-1 p
6. --- جنگلے دے سامݨے اک ٹھریا \VUgras{پھیری والا} \textsuperscript{(39)} اپݨے
\VUgras{ڈھوݨ دے کنڈیاں} \textsuperscript{(40)} تے اپݨی \VUgras{جُتے پالش دی پیٹی}
\textsuperscript{(41)} وچکار اُونگھ رہیا سی، جدوں کہ اک جوان \VUgras{دھوبݨ}
\textsuperscript{(42)}، کپڑیاں دی اک \VUgras{ٹوکری} \textsuperscript{(43)} چُکی، دروازے
کول کھڑے \VUgras{مکھ ویٹر} \textsuperscript{(44)} دی سنجیدہ شکل اُتے ہس رہی سی۔

%%K t13-07-1 p
7. --- \VUgras{رسوئیے} \textsuperscript{(45)} نوں ویکھ کے، جیہڑا اپݨی
\VUgras{رسوئی} \textsuperscript{(47)} توں، جیہڑی \VUgras{تہہ خانے} \textsuperscript{(48)}
وچ اے، اک \VUgras{رسوئی دے چھوکرے} \textsuperscript{(46)} نوں کم اُتے گھلݨ باہر آیا
سی، مینوں یاد آیا پئی کھاݨے دا ویلا کول اے، تے میں وِہڑا لنگھ کے کھاݨے دے گھر ول
گیا، جِتھے اک \VUgras{موٹر چالک} \textsuperscript{(50)} اپݨی \VUgras{موٹر}
\textsuperscript{(49)} کھڑی کر رہیا سی، جدوں کہ اک \VUgras{مستری} \textsuperscript{(52)}
\VUgras{سائیکل والے} \textsuperscript{(53)} دے دسے مطابق اک \VUgras{موٹر تِپہیا}
\textsuperscript{(51)} دی مرمت کر رہیا سی، جیدا ہُݨے ہُݨے حادثہ ہویا سی۔

%%K t13-08-1 p
8. --- اک جوان \VUgras{ویٹر} \textsuperscript{(54)} کولوں، جیہڑا \VUgras{بیئر دے
گلاس} \textsuperscript{(55)} چُکی سی، میں پُچھیا پئی سانجھی میز برامدے دے تھلے اے کہ
نئیں، تے اوہدے ہاں آکھݨ اُتے میں اک میز اُتے بہہ گیا تے میں اپݨے لئی اک ودھیا کھاݨا
منگوایا۔

%%K t13-09-1 p
9. --- بیرا میرے لئی دوپہر دے کھاݨے دا پرچہ (*) تے شراب دی سُوچی لیایا۔ میں چوݨ
کیتی تے \VUgras{پہلے کھاݨے} دے طور اُتے \VUgras{مکھݨ} نال \VUgras{جھینگے} منگائے،
تے پہلے وَرتاوے وچ \VUgras{کان دی وضع دی اوجھڑی}۔ میں \VUgras{مچھی} نہ لئی، پر
بھیڈ دی \VUgras{ران} دا اک حصہ منگیا تے \VUgras{ترکاری} وچ ملائی والی
\VUgras{پالک}۔ فیر، جدوں \VUgras{مِٹھے کھاݨیاں} وچوں کوئی مینوں نہ بھایا، میں کئی
طرحاں دیاں مِٹھیاں منگوائیاں : \VUgras{پنیر}، \VUgras{پھل} تے \VUgras{بسکٹ}۔ میں
نال اک ودھیا سینت ایمِلیوں \VUgras{شراب} جوڑی، تے اپݨے کھاݨے توں بہت تسلی پا کے
\VUgras{بیرے} \textsuperscript{(57)} نوں چنگی \VUgras{بخشش} \textsuperscript{(59)} دے
کے میز چھڈی۔

%%K t13-10-1 p
10. --- مینوں اُٹھݨ نوں تیار ویکھ کے \VUgras{منتظم} \textsuperscript{(60)} نے صلاح
دِتی پئی میں بالاخانے اُتے نکل آواں تے \VUgras{محیط} \textsuperscript{(62)} دا شاندار
منظر تکاں۔ دور نِکیاں مچھی دیاں \VUgras{کشتیاں} \textsuperscript{(63)}،
\VUgras{بادبانی جہاز} \textsuperscript{(64)} تے اک وڈا \VUgras{مسافر جہاز}
\textsuperscript{(65)} پچھاݨیا جا سکدا سی، جیدی کالی چھاں چٹے \VUgras{بدلاں}
\textsuperscript{(67)} اُتے اُبھردی سی، جیہڑے \VUgras{اُفق} \textsuperscript{(66)} اُتے
سن۔ اک ہولی ہوا اوس \VUgras{جھنڈے} \textsuperscript{(70)} نوں ہلا رہی سی جیہڑا
\VUgras{بورڈ} \textsuperscript{(69)} دے اُتے گڈیا اے، تے اوہ بورڈ \VUgras{ہال}
\textsuperscript{(68)} دا اے۔

11. --- منظر اِنّا دلچسپ سی پئی میں طے کیتا پئی اگلے دن کہوہ خانے دی چھت اُتے
چڑھاں گا، نال اک \VUgras{دوربین} \textsuperscript{(71)} یا اک \VUgras{دور درشی}
\textsuperscript{(72)} لے کے۔

%%K t13-12-1 p
12. --- بالاخانہ چھڈ کے میں ہوٹل دے کہوہ خانے وچ گیا، اک \VUgras{پیݨ دی چیز}
\textsuperscript{(73)} پیݨ تے \VUgras{بلیئرڈ} \textsuperscript{(75)} دی اک بازی کھیڈݨ۔
میں سِدھا کہوہ خانے دے کمرے وچوں لنگھیا، جِتھے بہت سارے گاہک \VUgras{شطرنج}
\textsuperscript{(78)}، \VUgras{چوپڑ} \textsuperscript{(79)}، \VUgras{ڈومینو}
\textsuperscript{(80)}، \VUgras{بیک گیمن} \textsuperscript{(81)} یا \VUgras{تاش}
\textsuperscript{(82)} کھیڈ رہے سن، تے سِدھا \VUgras{بلیئرڈ کمرے}
\textsuperscript{(74)} وچ جا اپڑیا۔

%%K t13-13-1 p
13. --- کھیڈ مُکݨ اُتے میں ہیٹھلی منزل اُتے اُتریا تے بیرے کولوں اک چِٹھی لکھݨ لئی
اک \VUgras{لفافہ} \textsuperscript{(84)}، کجھ \VUgras{کاغذ} \textsuperscript{(83)}، اک
\VUgras{ڈاک ٹکٹ} \textsuperscript{(85)}، اک \VUgras{قلم} \textsuperscript{(87)} تے کجھ
\VUgras{سیاہی} \textsuperscript{(86)} منگی۔

بھاء پُچھیا، تے سودا طے ہو جاوݨ اُتے اوہنے میرے لئی \VUgras{بگھی دا دروازہ}
